```rust
use band_core::BandError;
use band_core::ceremony::{CeremonyHttpRequest, OwnershipCeremony};
use band_core::events::{Event, Hand};
use band_core::identity::EnrollmentIdentity;
use band_core::responder::{Responder, SimMode};
use band_core::session::BandSession;
use p256::ecdsa::SigningKey;
use rand_core::OsRng;

/// Ping-pong host and band bytes until neither side has anything to say.
fn pump(session: &mut BandSession, band: &mut Responder, first: Vec<u8>, time: f64) -> band_core::Result<Vec<Event>> {
    let mut to_band = first;
    let mut events = Vec::new();
    for _ in 0..64 {
        if to_band.is_empty() {
            break;
        }
        let reply = band.respond(&to_band)?;
        let result = session.feed(&reply, time)?;
        events.extend(result.events);
        to_band = result.outgoing;
    }
    Ok(events)
}

fn start(session: &mut BandSession, band: &mut Responder) -> band_core::Result<Vec<Event>> {
    let request = session.request()?;
    pump(session, band, request, 100.0)
}

fn connected(events: &[Event]) -> bool {
    events.iter().any(|event| matches!(event, Event::Connected))
}
```

```rust
/// The band side of the transport handshake, driven from raw host bytes.
pub struct SimHandshake {
    transport: SecretKey,
    transport_point: Vec<u8>,
    band_challenge: Vec<u8>,
    band_seed: Vec<u8>,
    iv: [u8; 16],
    host_point: Vec<u8>,
    host_challenge: Vec<u8>,
}

impl SimHandshake {
    /// Answer the host's RequestEncryption frame (exactly one frame) with the
    /// band's RequestEncryption + EnableEncryption bytes.
    pub fn accept(host_request: &[u8]) -> Result<(SimHandshake, Vec<u8>)> {
        if host_request.len() < 12 {
            return Err(perr("sim: short RequestEncryption"));
        }
        let local = ProtoFields::parse(&host_request[12..])?;
        let host_point = local.bytes_len(1, 64)?.to_vec();
        let host_challenge = local.bytes_len(2, 16)?.to_vec();
        let mut scalar = [0u8; 32];
        scalar[31] = 1;
        let transport = SecretKey::from_slice(&scalar).expect("scalar 1 is a valid key");
        let transport_point = transport.public_key().to_encoded_point(false).as_bytes()[1..].to_vec();
        let band_challenge: Vec<u8> = (0..16).collect();
        let band_seed: Vec<u8> = (0..32).collect();
        let iv: [u8; 16] = std::array::from_fn(|i| 16 + i as u8);
        let peer_request = encode_frame(
            0x8001,
            &[0x81000005, 0x02000001],
            &[field_bytes(1, &transport_point), field_bytes(2, &band_challenge), field_int(3, 0), field_int(4, 3)].concat(),
        )?;
        let peer_enable = encode_frame(
            1,
            &[0x02000002],
            &[field_bytes(1, &transport_point), field_bytes(2, &band_seed), field_bytes(3, &iv), field_int(4, 42), field_int(5, 3)]
                .concat(),
        )?;
        let handshake = SimHandshake { transport, transport_point, band_challenge, band_seed, iv, host_point, host_challenge };
        Ok((handshake, [peer_request, peer_enable].concat()))
    }

    /// Take the host's EnableEncryption frame (exactly one frame) and derive both directions.
    pub fn finish(self, host_enable: &[u8]) -> Result<SimBand> {
        if host_enable.len() < 8 {
            return Err(perr("sim: short EnableEncryption"));
        }
        let enable = ProtoFields::parse(&host_enable[8..])?;
        if enable.bytes(1)? != self.host_point.as_slice() || enable.integer(5)? != 3 {
            return Err(perr("sim: unexpected EnableEncryption"));
        }
        let host_key = PublicKey::from_sec1_bytes(&[&[4u8][..], &self.host_point].concat())
            .map_err(|_| perr("sim: invalid host key"))?;
        let shared = ecdh::diffie_hellman(self.transport.to_nonzero_scalar(), host_key.as_affine());
        let host_seed = enable.bytes_len(2, 32)?.to_vec();
        let host_iv: [u8; 16] = enable.bytes_len(3, 16)?.try_into().expect("length checked");
        let host_counter = u32::try_from(enable.integer(4)?).map_err(|_| perr("sim: invalid counter"))?;
        Ok(SimBand {
            sender: AirShieldCipher::new(
                AirShieldKeys::derive(shared.raw_secret_bytes().as_slice(), &self.host_challenge, &self.band_seed)?,
                self.iv,
                42,
            ),
            receiver: AirShieldReceiver::new(AirShieldCipher::new(
                AirShieldKeys::derive(shared.raw_secret_bytes().as_slice(), &self.band_challenge, &host_seed)?,
                host_iv,
                host_counter,
            )),
            datax: DataXReceiver::default(),
            transport_point: self.transport_point,
            host_point: self.host_point,
            host_challenge: self.host_challenge,
            host_seed,
            band_challenge: self.band_challenge,
            band_seed: self.band_seed,
            stamp: 1_000_000,
        })
    }
}
```

`band-core/src/responder.rs`:
```rust
//! An auto-answering simulated band driven purely by bytes: the band side of a
//! whole connection (handshake, trust, ceremony, subscription, hand, battery,
//! raw sEMG, stop) for daemon tests and `kinesisd --dry-run`. Nothing is secret.

use p256::ecdsa::SigningKey;

use crate::datax::{DataXFrame, be16, encode_frame};
use crate::error::Result;
use crate::identity::point64;
use crate::proto::{ProtoFields, field_bytes, field_int};
use crate::sim::{SimBand, SimHandshake};

pub const SIM_SERIAL: &str = "SIMBAND0001";
const RPC: u32 = 0x02000315;
const DEVICE_RECEIPT: &str = r#"{"receipt_type":"DevicePendingOwnershipReceipt","additional_data":"{}"}"#;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SimMode {
    /// A band in pairing mode answering the legacy startup.
    Legacy,
    /// A band owned by this host: answers EnableTrust with its identity proof.
    Enrolled,
    /// A band in pairing mode that runs the ownership ceremony, then trusts the new owner.
    Ceremony,
}

enum State {
    Request,
    Enable(SimHandshake),
    Ready(SimBand),
    Broken,
}

pub struct Responder {
    mode: SimMode,
    state: State,
    pending: Vec<u8>,
    band_key: SigningKey,
    /// Answer EnableTrust with this result word instead of success (e.g. 0x03001043).
    pub reject_identity: Option<u32>,
    /// Answer StartChangeOwner with this result word (e.g. 0x03001042, wrong account).
    pub reject_claim: Option<u32>,
    /// ConfigResp field 10: 0 = right, 1 = left.
    pub hand: u64,
    /// (level, charging flag).
    pub battery: (u64, Option<u64>),
    /// Stop answering (the band went quiet); requests are still decrypted.
    pub silent: bool,
    streams: u64,
    raw: u64,
    sequence: u64,
    finish_seen: bool,
    stopped: bool,
}

fn frame_size(buffer: &[u8]) -> Option<usize> {
    if buffer.len() < 4 {
        return None;
    }
    let size = usize::from(be16(buffer, 0) & 0x7fff) + 4;
    (buffer.len() >= size).then_some(size)
}

fn reply(band: &mut SimBand, channel: u16, word: u32, payload: &[u8]) -> Result<Vec<u8>> {
    band.encrypt(&encode_frame(channel, &[word], payload)?)
}

impl Responder {
    pub fn new(mode: SimMode, band_key: SigningKey) -> Self {
        Self {
            mode,
            state: State::Request,
            pending: Vec::new(),
            band_key,
            reject_identity: None,
            reject_claim: None,
            hand: 0,
            battery: (80, Some(0)),
            silent: false,
            streams: 0,
            raw: 0,
            sequence: 0,
            finish_seen: false,
            stopped: false,
        }
    }

    /// The band identity key's 64-byte point (what the ownership server reports).
    pub fn band_point(&self) -> Vec<u8> {
        point64(self.band_key.verifying_key())
    }

    /// True once FinishChangeOwner reached the band.
    pub fn finish_seen(&self) -> bool {
        self.finish_seen
    }

    /// True once the band acknowledged a stop request.
    pub fn stopped(&self) -> bool {
        self.stopped
    }

    /// Consume host bytes and return every reply the band sends.
    pub fn respond(&mut self, bytes: &[u8]) -> Result<Vec<u8>> {
        self.pending.extend_from_slice(bytes);
        let mut out = Vec::new();
        loop {
            match std::mem::replace(&mut self.state, State::Broken) {
                State::Request => {
                    let Some(size) = frame_size(&self.pending) else {
                        self.state = State::Request;
                        break;
                    };
                    let request: Vec<u8> = self.pending.drain(..size).collect();
                    let (handshake, answer) = SimHandshake::accept(&request)?;
                    out.extend(answer);
                    self.state = State::Enable(handshake);
                }
                State::Enable(handshake) => {
                    let Some(size) = frame_size(&self.pending) else {
                        self.state = State::Enable(handshake);
                        break;
                    };
                    let enable: Vec<u8> = self.pending.drain(..size).collect();
                    self.state = State::Ready(handshake.finish(&enable)?);
                }
                State::Ready(mut band) => {
                    let bytes = std::mem::take(&mut self.pending);
                    let frames = band.requests(&bytes);
                    let result = frames.and_then(|frames| {
                        let mut answers = Vec::new();
                        for frame in &frames {
                            answers.extend(self.answer(&mut band, frame)?);
                        }
                        Ok(answers)
                    });
                    self.state = State::Ready(band);
                    out.extend(result?);
                    break;
                }
                State::Broken => return Err(crate::error::BandError::Protocol("sim: responder is broken".into())),
            }
        }
        Ok(if self.silent { Vec::new() } else { out })
    }

    fn ready(&mut self) -> Result<&mut SimBand> {
        match &mut self.state {
            State::Ready(band) => Ok(band),
            _ => Err(crate::error::BandError::Protocol("sim: not connected yet".into())),
        }
    }

    /// A recognized-gesture message (finger/action/derived use the band's enum values).
    pub fn gesture(&mut self, action: u64, finger: u64, derived: u64) -> Result<Vec<u8>> {
        self.sequence += 1;
        let sequence = self.sequence;
        let band = self.ready()?;
        band.stamp += 10_000;
        let payload = [
            field_int(1, sequence),
            field_int(2, band.stamp),
            field_int(3, finger),
            field_int(4, action),
            field_int(5, derived),
        ]
        .concat();
        band.encrypt(&encode_frame(0x8010, &[0x0200020d], &payload)?)
    }

    /// One gyro sample 10 ms after the previous one.
    pub fn gyro(&mut self, x: i16) -> Result<Vec<u8>> {
        let band = self.ready()?;
        band.stamp += 10_000;
        let axes = [x.to_le_bytes(), 0i16.to_le_bytes(), 0i16.to_le_bytes()].concat();
        let payload = [field_int(1, band.stamp), field_int(2, band.stamp), field_bytes(3, &axes)].concat();
        band.encrypt(&encode_frame(0x8010, &[0x0200020f], &payload)?)
    }

    /// One raw sEMG batch (256 bytes of mid-scale samples).
    pub fn raw_emg(&mut self) -> Result<Vec<u8>> {
        self.sequence += 1;
        let sequence = self.sequence;
        let band = self.ready()?;
        band.stamp += 7_813;
        let payload = [field_int(1, sequence), field_int(2, band.stamp), field_bytes(3, &[0x80; 256])].concat();
        band.encrypt(&encode_frame(5, &[0x0200020a], &payload)?)
    }

    fn answer(&mut self, band: &mut SimBand, frame: &DataXFrame) -> Result<Vec<u8>> {
        let kind = frame.words.last().copied();
        match (frame.channel, kind) {
            (0x8002, Some(0x02003000)) if self.mode == SimMode::Ceremony => {
                let payload = [field_bytes(1, &[1; 967]), field_bytes(2, SIM_SERIAL.as_bytes()), field_bytes(5, &[2; 889])].concat();
                reply(band, 2, 0x02003001, &payload)
            }
            // The legacy empty identity query needs no answer.
            (0x8002, Some(0x02003000)) => Ok(Vec::new()),
            (0x8002, Some(0x02002000)) => {
                let nonce: Vec<u8> = (0..16).collect();
                reply(band, 2, 0x02002001, &field_bytes(1, &nonce))
            }
            (0x8002, Some(0x02002002)) => match self.reject_claim {
                Some(code) => reply(band, 2, code, &[]),
                None => {
                    let signature = [vec![0x30, 0x44], vec![4; 68]].concat();
                    reply(band, 2, 0x02002003, &[field_bytes(1, &signature), field_bytes(2, DEVICE_RECEIPT.as_bytes())].concat())
                }
            },
            (0x8002, Some(0x02002004)) => {
                self.finish_seen = true;
                self.mode = SimMode::Enrolled;
                reply(band, 2, 0x02002005, &[])
            }
            (0x8002, Some(0x02001000)) => {
                let result = self.reject_identity.unwrap_or(0x03001000);
                let mut out = reply(band, 2, result, &[])?;
                if result == 0x03001000 {
                    let proof = band.trust_proof(&self.band_key)?;
                    out.extend(band.encrypt(&proof)?);
                }
                Ok(out)
            }
            (0x8001, Some(0x02001000)) => {
                reply(band, 0x8001, 0x02001000, &[field_int(1, 1), field_bytes(2, &[1; 16])].concat())
            }
            (0x8003, _) => reply(band, 3, RPC, &[field_int(1, 1), field_int(2, 1), field_bytes(4, &[])].concat()),
            (0x8005, _) => self.stream(band, frame),
            (0x8006, _) => {
                let fields = ProtoFields::parse(&frame.payload)?;
                let config = ProtoFields::parse(fields.bytes(5).unwrap_or(&[]))?;
                if config.contains(10) {
                    self.hand = config.integer(10)?;
                }
                let answer = [field_int(10, self.hand), field_int(2, 2048)].concat();
                reply(band, 6, RPC, &[field_int(1, fields.integer(1)?), field_int(2, 1), field_bytes(6, &answer)].concat())
            }
            (0x8007, _) => {
                let id = ProtoFields::parse(&frame.payload)?.integer(1)?;
                let emg = [field_int(1, 2048), field_int(2, 8), field_int(4, 16), field_int(5, 16), field_int(10, 0)].concat();
                reply(band, 7, RPC, &[field_int(1, id), field_int(2, 1), field_bytes(6, &field_bytes(42, &emg))].concat())
            }
            (0x8008, _) => {
                let id = ProtoFields::parse(&frame.payload)?.integer(1)?;
                let (level, charging) = self.battery;
                let mut battery = field_int(1, level);
                if let Some(charging) = charging {
                    battery.extend(field_int(2, charging));
                }
                reply(band, 8, RPC, &[field_int(1, id), field_int(2, 1), field_bytes(3, &field_bytes(1, &battery))].concat())
            }
            _ => Ok(Vec::new()),
        }
    }

    /// The input subscription channel: open (id 2, no reply), enable/update
    /// (apply the control flags), status query (empty control), stop (id 4).
    fn stream(&mut self, band: &mut SimBand, frame: &DataXFrame) -> Result<Vec<u8>> {
        let fields = ProtoFields::parse(&frame.payload)?;
        let id = fields.integer(1)?;
        if id == 2 {
            return Ok(Vec::new());
        }
        let control = ProtoFields::parse(fields.bytes(4).unwrap_or(&[]))?;
        if control.contains(2) {
            self.raw = control.integer(2)?;
        }
        if control.contains(3) {
            self.streams = control.integer(3)?;
        }
        if id == 4 && self.streams == 0 {
            self.stopped = true;
        }
        let flags = [field_int(2, self.raw), field_int(3, self.streams), field_int(6, self.streams), field_int(8, self.streams)].concat();
        reply(band, 5, RPC, &[field_int(1, id), field_int(2, 1), field_bytes(5, &flags)].concat())
    }
}
```

```rust
        if !(0.5..=4.0).contains(&config.dial.sensitivity) {
            return Err("dial.sensitivity must be between 0.5 and 4".into());

    pub fn load(path: &Path) -> Result<Self, String> {

            Ok(text) => Self::parse(&text).map_err(|error| format!("{}: {error}", path.display())),
            Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
            Err(error) => Err(format!("{}: {error}", path.display())),
        }

            .map(PathBuf::from)
            .unwrap_or_else(|| home().join(".config"));
        base.join("kinesis").join("config.toml")
    }
```

```rust
const BAND: &str = "band.json";
pub(crate) const META_SESSION: &str = "meta-session.json";

/// The paired band: its Bluetooth address and advertised name.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct BandRecord {
    pub address: String,
    pub name: String,
}

#[derive(Debug, Clone)]
pub struct Store {
    dir: PathBuf,
}

impl Store {
    pub fn open(dir: impl Into<PathBuf>) -> io::Result<Self> {
        let dir = dir.into();
        DirBuilder::new().recursive(true).mode(0o700).create(&dir)?;
        fs::set_permissions(&dir, fs::Permissions::from_mode(0o700))?;
        Ok(Self { dir })
    }

    /// `$XDG_STATE_HOME/kinesis`, else `~/.local/state/kinesis`.
    pub fn default_dir() -> PathBuf {
        std::env::var_os("XDG_STATE_HOME")
            .map(PathBuf::from)
            .unwrap_or_else(|| crate::config::home().join(".local/state"))
            .join("kinesis")
    }

    pub fn dir(&self) -> &Path {
        &self.dir
    }

    fn path(&self, name: &str) -> PathBuf {
        self.dir.join(name)
    }

    pub(crate) fn write_secret(&self, name: &str, bytes: &[u8]) -> io::Result<()> {
        let temp = self.path(&format!(".{name}.tmp"));
        {
            let mut file = OpenOptions::new().write(true).create(true).truncate(true).mode(0o600).open(&temp)?;
            file.write_all(bytes)?;
            file.sync_all()?;
        }
        fs::rename(&temp, self.path(name))
    }

    pub(crate) fn read(&self, name: &str) -> io::Result<Option<Vec<u8>>> {
        match fs::read(self.path(name)) {
            Ok(bytes) => Ok(Some(bytes)),
            Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(None),
            Err(error) => Err(error),
        }
    }

    pub(crate) fn remove(&self, name: &str) -> io::Result<()> {
        match fs::remove_file(self.path(name)) {
            Err(error) if error.kind() != io::ErrorKind::NotFound => Err(error),
            _ => Ok(()),
        }
    }

    fn write_identity(&self, owner: &str, band_key: &str, identity: &EnrollmentIdentity) -> io::Result<()> {
        self.write_secret(owner, &identity.private_bytes())?;
        match identity.band_key_x963() {
            Some(key) => self.write_secret(band_key, &key),
            None => self.remove(band_key),
        }
    }

    fn read_identity(&self, owner: &str, band_key: &str) -> io::Result<Option<EnrollmentIdentity>> {
        let Some(private) = self.read(owner)? else { return Ok(None) };
        let band = self.read(band_key)?;
        EnrollmentIdentity::from_bytes(&private, band.as_deref())
            .map(Some)
            .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, format!("{owner}: {error}")))
    }

    /// Store the confirmed owner identity and drop any pending one.
    pub fn save_identity(&self, identity: &EnrollmentIdentity) -> io::Result<()> {
        self.write_identity(OWNER, BAND_KEY, identity)?;
        self.remove(PENDING_OWNER)?;
        self.remove(PENDING_BAND_KEY)
    }

    pub fn load_identity(&self) -> io::Result<Option<EnrollmentIdentity>> {
        self.read_identity(OWNER, BAND_KEY)
    }

    /// Store the identity the band may commit to before its confirmation arrives.
    pub fn save_pending(&self, identity: &EnrollmentIdentity) -> io::Result<()> {
        self.write_identity(PENDING_OWNER, PENDING_BAND_KEY, identity)
    }

    pub fn load_pending(&self) -> io::Result<Option<EnrollmentIdentity>> {
        self.read_identity(PENDING_OWNER, PENDING_BAND_KEY)
    }

    /// The identity to connect with: the confirmed one, else a pending one
    /// (the band may have committed a claim whose confirmation was lost).
    pub fn connect_identity(&self) -> io::Result<Option<EnrollmentIdentity>> {
        match self.load_identity()? {
            Some(identity) => Ok(Some(identity)),
            None => self.load_pending(),
        }
    }

    pub fn save_band(&self, band: &BandRecord) -> io::Result<()> {
        let json = serde_json::to_vec_pretty(band).map_err(io::Error::other)?;
        self.write_secret(BAND, &json)
    }

    pub fn load_band(&self) -> io::Result<Option<BandRecord>> {
        match self.read(BAND)? {
            Some(bytes) => serde_json::from_slice(&bytes)
                .map(Some)
                .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, format!("{BAND}: {error}"))),
            None => Ok(None),
        }
    }

    /// Remove every stored key, the band record and the Meta session.
    pub fn forget(&self) -> io::Result<()> {
        for name in [OWNER, BAND_KEY, PENDING_OWNER, PENDING_BAND_KEY, BAND, META_SESSION] {
            self.remove(name)?;
        }
        Ok(())
    }
}
```

```rust
//! What to do with each band event — a port of the event handling in kinesis
//! `BandModel.swift` (~L840–990). Pure: no I/O; time is passed in.

use band_core::events::{Event, Hand};
use band_core::gestures::{ActionGate, DialRouter, GestureRouter, Recognized, Swipe, Tap};

use crate::config::{Config, DialTarget, HandSetting};
use crate::status::Status;

#[derive(Debug, Clone, PartialEq)]
pub enum Command {
    /// Run this shell command.
    Run(String),
    /// Ask the band to switch hands.
    SetHand(Hand),
}

fn label(gesture: Recognized) -> &'static str {
    match gesture {
        Recognized::Swipe(Swipe::Left) => "swipe left",
        Recognized::Swipe(Swipe::Right) => "swipe right",
        Recognized::Swipe(Swipe::Up) => "swipe up",
        Recognized::Swipe(Swipe::Down) => "swipe down",
        Recognized::Tap(Tap::IndexTap) => "index tap",
        Recognized::Tap(Tap::IndexDoubleTap) => "index double tap",
        Recognized::Tap(Tap::MiddleTap) => "middle tap",
        Recognized::Tap(Tap::MiddleDoubleTap) => "middle double tap",
        Recognized::Tap(Tap::MiddleHold) => "middle hold",
    }
}

fn hand_name(hand: Hand) -> &'static str {
    match hand {
        Hand::Left => "left",
        Hand::Right => "right",
    }
}

pub struct Controller {
    config: Config,
    dial_target: DialTarget,
    router: GestureRouter,
    gate: ActionGate,
    dial_gate: ActionGate,
    dial_router: DialRouter,
    /// Controls are on (the user hasn't paused).
    enabled: bool,
    /// The band's input subscription is up.
    live: bool,
    hand: Option<Hand>,
    hand_confirmed: bool,
    hand_requested: bool,
    dial_armed: bool,
    dial_engaged: bool,
    dial_turned: bool,
    dial_ended_at: f64,
    last_dial_action: f64,
    status: Status,
}

impl Controller {
    pub fn new(config: Config) -> Self {
        let enabled = config.band.autostart;
        let status = Status { paused: !enabled, dial: config.dial.target, ..Status::default() };
        Self {
            dial_target: config.dial.target,
            config,
            router: GestureRouter::default(),
            gate: ActionGate::default(),
            dial_gate: ActionGate::new(0.0),
            dial_router: DialRouter::default(),
            enabled,
            live: false,
            hand: None,
            hand_confirmed: false,
            hand_requested: false,
            dial_armed: false,
            dial_engaged: false,
            dial_turned: false,
            dial_ended_at: f64::NEG_INFINITY,
            last_dial_action: f64::NEG_INFINITY,
            status,
        }
    }

    /// Gesture, dial and control fields of the status (the daemon adds the rest).
    pub fn status(&self) -> &Status {
        &self.status
    }

    /// Apply a reloaded config. A changed dial target in the file wins over a
    /// runtime switch; a changed hand setting is applied if the band is live.
    pub fn set_config(&mut self, config: Config) -> Vec<Command> {
        if config.dial.target != self.config.dial.target {
            self.dial_target = config.dial.target;
            self.status.dial = config.dial.target;
        }
        self.config = config;
        self.hand_requested = false;
        self.reset_dial();
        self.maybe_request_hand()
    }

    pub fn resume(&mut self, now: f64) {
        self.enabled = true;
        self.status.paused = false;
        if self.live {
            self.gate.arm(now);
        }
        self.reset_dial();
    }

    pub fn pause(&mut self) {
        self.enabled = false;
        self.status.paused = true;
        self.suspend();
    }

    pub fn toggle(&mut self, now: f64) {
        if self.enabled { self.pause() } else { self.resume(now) }
    }

    pub fn set_dial_target(&mut self, target: DialTarget) {
        self.dial_target = target;
        self.status.dial = target;
        self.reset_dial();
    }

    pub fn cycle_dial(&mut self) {
        self.set_dial_target(self.dial_target.next());
    }

    pub fn on_disconnect(&mut self) {
        self.live = false;
        self.hand_confirmed = false;
        self.hand_requested = false;
        self.suspend();
        self.router.reset();
        self.status.charging = None;
    }

    pub fn on_event(&mut self, event: &Event, now: f64) -> Vec<Command> {
        match event {
            Event::Connected => {
                self.live = true;
                if self.enabled {
                    self.gate.arm(now);
                }
                self.maybe_request_hand()
            }
            Event::Handedness(hand) => {
                self.hand = Some(*hand);
                self.hand_confirmed = true;
                self.status.hand = Some(hand_name(*hand).into());
                self.maybe_request_hand()
            }
            Event::HandednessFailure(_) => {
                self.hand_confirmed = false;
                Vec::new()
            }
            Event::BatteryStatus(Some(status)) => {
                self.status.battery = Some(status.level);
                self.status.charging = status.charging;
                Vec::new()
            }
            Event::BatteryStatus(None) => {
                self.status.charging = None;
                Vec::new()
            }
            Event::Gesture(message) => self.gesture(message, now),
            Event::DialState(engaged) => {
                self.dial_state(*engaged, now);
                Vec::new()
            }
            Event::DialTurn(rotation) => self.dial_turn(*rotation, now),
            _ => Vec::new(),
        }
    }

    fn gesture(&mut self, message: &band_core::events::GestureMessage, now: f64) -> Vec<Command> {
        let age = now - message.received_at;
        if !(-0.1..=0.35).contains(&age) {
            return Vec::new();
        }
        let Some(gesture) = self.router.gesture(message, now) else { return Vec::new() };
        // Releasing a wrist turn must not also trigger an index-finger assignment.
        if let Recognized::Tap(tap) = gesture
            && tap.finger() == "index"
            && (now - self.last_dial_action <= 0.6 || now - self.dial_ended_at <= 0.7)
        {
            return Vec::new();
        }
        self.status.gestures += 1;
        self.status.last_gesture = Some(label(gesture).into());
        let command = self.config.gestures.command(gesture).to_owned();
        if !self.enabled || command.is_empty() || !self.gate.allows(message.received_at, now, self.live, true) {
            return Vec::new();
        }
        self.status.last_action = Some(command.clone());
        vec![Command::Run(command)]
    }

    fn dial_state(&mut self, engaged: bool, now: f64) {
        if !self.live {
            self.reset_dial();
            return;
        }
        // A pinch that turned has just ended: its release is not a tap.
        if !engaged && self.dial_turned {
            self.dial_ended_at = now;
        }
        self.dial_turned = false;
        self.dial_engaged = engaged;
        self.reset_dial();
        if engaged && self.enabled {
            self.dial_armed = true;
            self.dial_gate.arm(now);
        }
    }

    fn dial_turn(&mut self, rotation: f64, now: f64) -> Vec<Command> {
        if !self.live || !self.hand_confirmed || !self.dial_engaged || !rotation.is_finite() {
            return Vec::new();
        }
        // The same intended turn produces the opposite gyro sign on the left wrist.
        let delta = if self.hand == Some(Hand::Left) { -rotation } else { rotation };
        self.dial_turned = true;
        if !self.enabled || !self.dial_armed || self.config.dial.command(self.dial_target, true).is_none() {
            return Vec::new();
        }
        if !self.dial_gate.allows(now, now, self.live, true) {
            return Vec::new();
        }
        let steps = self.dial_router.turn(delta, self.config.dial.sensitivity, now);
        if steps == 0 {
            return Vec::new();
        }
        self.last_dial_action = now;
        match self.config.dial.command(self.dial_target, steps > 0) {
            Some(command) => {
                self.status.last_action = Some(command.to_owned());
                vec![Command::Run(command.to_owned())]
            }
            None => Vec::new(),
        }
    }

    fn desired_hand(&self) -> Option<Hand> {
        match self.config.band.hand {
            HandSetting::Band => None,
            HandSetting::Left => Some(Hand::Left),
            HandSetting::Right => Some(Hand::Right),
        }
    }

    fn maybe_request_hand(&mut self) -> Vec<Command> {
        match (self.desired_hand(), self.hand) {
            (Some(wanted), Some(current)) if self.live && wanted != current && !self.hand_requested => {
                self.hand_requested = true;
                vec![Command::SetHand(wanted)]
            }
            _ => Vec::new(),
        }
    }

    fn reset_dial(&mut self) {
        self.dial_armed = false;
        self.dial_gate.pause();
        self.dial_router.reset();
    }

    fn suspend(&mut self) {
        self.gate.pause();
        self.reset_dial();
        self.dial_engaged = false;
    }
}
```

```rust
//! Running configured commands: detached, through `sh -c`, killed after a
//! timeout so a stuck command can never back up the band connection.

use std::process::Stdio;
use std::sync::{Arc, Mutex};
use std::time::Duration;

pub trait Runner: Send + Sync + 'static {
    /// Start `command` and return immediately.
    fn run(&self, command: String);
}

#[derive(Debug, Clone)]
pub struct ShellRunner {
    pub timeout: Duration,
}

impl Default for ShellRunner {
    fn default() -> Self {
        Self { timeout: Duration::from_secs(5) }
    }
}

impl Runner for ShellRunner {
    fn run(&self, command: String) {
        let timeout = self.timeout;
        tokio::spawn(async move {
            let child = tokio::process::Command::new("sh")
                .arg("-c")
                .arg(&command)
                .stdin(Stdio::null())
                .stdout(Stdio::null())
                .stderr(Stdio::null())
                .kill_on_drop(true)
                .spawn();
            let mut child = match child {
                Ok(child) => child,
                Err(error) => {
                    tracing::warn!(%command, "could not start action: {error}");
                    return;
                }
            };
            match tokio::time::timeout(timeout, child.wait()).await {
                Ok(Ok(status)) if !status.success() => tracing::warn!(%command, %status, "action failed"),
                Ok(Err(error)) => tracing::warn!(%command, "action wait failed: {error}"),
                Err(_) => {
                    let _ = child.kill().await;
                    tracing::warn!(%command, "action timed out and was killed");
                }
                Ok(Ok(_)) => {}
            }
        });
    }
}

/// Records commands instead of running them (tests, `--dry-run` logs).
#[derive(Debug, Clone, Default)]
pub struct RecordingRunner(Arc<Mutex<Vec<String>>>);

impl RecordingRunner {
    pub fn commands(&self) -> Vec<String> {
        self.0.lock().expect("runner lock").clone()
    }
}

impl Runner for RecordingRunner {
    fn run(&self, command: String) {
        self.0.lock().expect("runner lock").push(command);
    }
}
```

```rust
struct SimState {
    mode: SimMode,
    next_mode: Option<SimMode>,
    band_key: SigningKey,
    reject_identity: Option<u32>,
    reject_claim: Option<u32>,
    /// SHA-256 of the owner's app point; kept across connections like a real band.
    owner: Option<[u8; 32]>,
    /// ConfigResp hand (0 right, 1 left); kept across connections.
    hand: u64,
    inject: Option<mpsc::UnboundedSender<SimInject>>,
    opens: usize,
}

pub struct SimTransport {
    state: Arc<Mutex<SimState>>,
}

impl SimTransport {
    /// `mode` is how the band answers ordinary connections. `prepare_pairing`
    /// switches the next connection to the ceremony; afterwards the band is enrolled.
    pub fn new(mode: SimMode) -> Self {
        let mut scalar = [0u8; 32];
        scalar[31] = 2;
        let band_key = SigningKey::from_slice(&scalar).expect("scalar 2 is a valid key");
        Self {
            state: Arc::new(Mutex::new(SimState {
                mode,
                next_mode: None,
                band_key,
                reject_identity: None,
                reject_claim: None,
                owner: None,
                hand: 0,
                inject: None,
                opens: 0,
            })),
        }
    }

    fn lock(&self) -> std::sync::MutexGuard<'_, SimState> {
        self.state.lock().expect("sim transport lock")
    }

    /// The simulated band's identity key point (what the fake ownership server returns).
    pub fn band_point(&self) -> Vec<u8> {
        point64(self.lock().band_key.verifying_key())
    }

    pub fn set_reject_identity(&self, code: Option<u32>) {
        self.lock().reject_identity = code;
    }

    pub fn set_reject_claim(&self, code: Option<u32>) {
        self.lock().reject_claim = code;
    }

    /// Make the band already owned by `identity` (as if a claim had happened).
    pub fn set_owner(&self, identity: &EnrollmentIdentity) {
        let hash: [u8; 32] = Sha256::digest(identity.app_point()).into();
        self.lock().owner = Some(hash);
    }

    /// The band's current owner hash (None until a claim or the first enrolled connection).
    pub fn owner(&self) -> Option<[u8; 32]> {
        self.lock().owner
    }

    /// Act on the most recently opened connection; false if there is none.
    pub fn inject(&self, what: SimInject) -> bool {
        self.lock().inject.as_ref().is_some_and(|sender| sender.send(what).is_ok())
    }

    /// How many connections have been opened.
    pub fn opens(&self) -> usize {
        self.lock().opens
    }
}

impl Transport for SimTransport {
    type Io = DuplexStream;
    type PairingGuard = ();

    async fn scan(&self, _timeout: Duration) -> Result<Vec<Found>, String> {
        Ok(vec![Found { address: SIM_ADDRESS.into(), name: SIM_NAME.into(), rssi: Some(-40) }])
    }

    async fn prepare_pairing(&self, _address: &str) -> Result<(), String> {
        self.lock().next_mode = Some(SimMode::Ceremony);
        Ok(())
    }

    async fn open(&self, address: &str) -> Result<DuplexStream, String> {
        if address != SIM_ADDRESS {
            return Err(format!("no simulated band at {address}"));
        }
        let (host, band) = tokio::io::duplex(1 << 16);
        let (sender, receiver) = mpsc::unbounded_channel();
        let responder = {
            let mut state = self.lock();
            let mode = state.next_mode.take().unwrap_or(state.mode);
            if mode == SimMode::Ceremony {
                state.mode = SimMode::Enrolled;
            }
            state.inject = Some(sender);
            state.opens += 1;
            let mut responder = Responder::new(mode, state.band_key.clone());
            responder.reject_identity = state.reject_identity;
            responder.reject_claim = state.reject_claim;
            responder.set_owner(state.owner);
            responder.hand = state.hand;
            responder
        };
        tokio::spawn(serve(band, responder, receiver, self.state.clone()));
        Ok(host)
    }

    async fn forget(&self, _address: &str) -> Result<(), String> {
        Ok(())
    }
}

async fn serve(
    mut io: DuplexStream,
    mut band: Responder,
    mut inject: mpsc::UnboundedReceiver<SimInject>,
    state: Arc<Mutex<SimState>>,
) {
    let mut buffer = vec![0u8; 1 << 16];
    loop {
        // What a real band remembers survives the connection.
        {
            let mut state = state.lock().expect("sim transport lock");
            state.owner = band.owner();
            state.hand = band.hand;
        }
        tokio::select! {
            read = io.read(&mut buffer) => {
                let count = match read {
                    Ok(0) | Err(_) => return,
                    Ok(count) => count,
                };
                match band.respond(&buffer[..count]) {
                    Ok(reply) if !reply.is_empty() => {
                        if io.write_all(&reply).await.is_err() {
                            return;
                        }
                    }
                    Ok(_) => {}
                    Err(_) => return,
                }
            }
            what = inject.recv() => {
                let bytes = match what {
                    None | Some(SimInject::Close) => return,
                    Some(SimInject::Silence) => {
                        band.silent = true;
                        continue;
                    }
                    Some(SimInject::Gesture { action, finger, derived }) => band.gesture(action, finger, derived),
                    Some(SimInject::Gyro(x)) => band.gyro(x),
                    Some(SimInject::RawEmg) => band.raw_emg(),
                };
                if let Ok(bytes) = bytes
                    && io.write_all(&bytes).await.is_err()
                {
                    return;
                }
            }
        }
    }
}
```

```rust
//! (kinesis `BandConnection.swift` scheduling + `BandModel.swift` stall check).

use band_core::BandError;
use band_core::events::{Event, Hand};
use band_core::identity::EnrollmentIdentity;
use band_core::session::BandSession;
use tokio::io::{AsyncRead, AsyncReadExt, AsyncWrite, AsyncWriteExt};
use tokio::sync::mpsc;

use crate::clock::Clock;

const TICK: Duration = Duration::from_millis(50);
const STARTUP_TIMEOUT: f64 = 30.0;
/// The ownership ceremony includes two HTTP round trips.
const CEREMONY_TIMEOUT: f64 = 180.0;
const BATTERY_EVERY: f64 = 5.0;
const QUIET_QUERY: f64 = 2.0;
const STALL_LIVE: f64 = 8.0;
const STALL_BEFORE_INPUT: f64 = 50.0;
const STOP_WAIT: f64 = 3.0;

#[derive(Debug)]
pub enum LinkCommand {
    /// Stop the streams and close (also what dropping the sender does).
    Stop,
    SetHand(Hand),
    RawEmg(bool),
    /// The `pair_request` HTTP reply.
    ResumePairRequest { signature: Vec<u8>, receipt: String },
    /// The `pair` HTTP reply.
    ResumePair { signature: Vec<u8>, receipt: String, device_public_key: Option<Vec<u8>> },
}

#[derive(Debug)]
pub enum LinkEvent {
    Band(Event),
    /// A command the session refused for now (`BandError::Busy`); the link is fine.
    Refused(String),
}

#[derive(Debug, Clone, PartialEq)]
pub enum LinkEnd {
    /// Stopped on request (acknowledged or not).
    Closed,
    /// Went quiet for too long; stopped.
    Stalled,
    /// Didn't finish starting up in time.
    StartupTimeout,
    /// The band closed the channel.
    StreamEnded,
    /// The band rejected our stored owner key.
    IdentityRejected(String),
    /// The band rejected the ownership claim (`0x1042` = a different Meta account).
    OwnershipRejected { code: u32, message: String },
    Failed(String),
}

impl std::fmt::Display for LinkEnd {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            LinkEnd::Closed => f.write_str("disconnected"),
            LinkEnd::Stalled => f.write_str("the band stopped responding"),
            LinkEnd::StartupTimeout => f.write_str("the band took too long to respond"),
            LinkEnd::StreamEnded => f.write_str("the band closed the connection"),
            LinkEnd::IdentityRejected(message) | LinkEnd::Failed(message) => f.write_str(message),
            LinkEnd::OwnershipRejected { message, .. } => f.write_str(message),
        }
    }
}

pub struct LinkOutcome {
    pub end: LinkEnd,
    /// The session's owner identity at the end — after a claim, the new one,
    /// even if the connection failed right after the band confirmed it.
    pub enrollment: Option<EnrollmentIdentity>,
}

fn from_error(error: BandError) -> LinkEnd {
    match error {
        BandError::IdentityMismatch(message) => LinkEnd::IdentityRejected(message),
        BandError::OwnershipRejected { code, message } => LinkEnd::OwnershipRejected { code, message },
        BandError::Protocol(message) | BandError::Busy(message) => LinkEnd::Failed(message),
    }
}

async fn send<IO: AsyncWrite + Unpin>(io: &mut IO, bytes: &[u8]) -> Result<(), LinkEnd> {
    if bytes.is_empty() {
        return Ok(());
    }
    io.write_all(bytes).await.map_err(|error| LinkEnd::Failed(format!("write to the band failed: {error}")))?;
    io.flush().await.map_err(|error| LinkEnd::Failed(format!("write to the band failed: {error}")))
}

/// Run one connection until it ends. `persist_pending` is called with the
/// claimed identity before FinishChangeOwner is written; if it fails the
/// claim is abandoned before the band can commit.
pub async fn run_link<IO>(
    mut io: IO,
    mut session: BandSession,
    clock: Clock,
    mut commands: mpsc::Receiver<LinkCommand>,
    events: mpsc::Sender<LinkEvent>,
    mut persist_pending: impl FnMut(&EnrollmentIdentity) -> std::io::Result<()>,
) -> LinkOutcome
where
    IO: AsyncRead + AsyncWrite + Unpin,
{
    let end = drive(&mut io, &mut session, clock, &mut commands, &events, &mut persist_pending).await;
    let _ = io.shutdown().await;
    LinkOutcome { end, enrollment: session.enrollment().cloned() }
}

struct Stopping {
    until: f64,
    reason: LinkEnd,
}

async fn drive<IO: AsyncRead + AsyncWrite + Unpin>(
    io: &mut IO,
    session: &mut BandSession,
    clock: Clock,
    commands: &mut mpsc::Receiver<LinkCommand>,
    events: &mpsc::Sender<LinkEvent>,
    persist_pending: &mut impl FnMut(&EnrollmentIdentity) -> std::io::Result<()>,
) -> LinkEnd {
    let start = clock.now();
    let ceremony = session.ceremony().is_some();
    let mut deadline = start + if ceremony { CEREMONY_TIMEOUT } else { STARTUP_TIMEOUT };
    let mut live = false;
    let mut last_read = start;
    let mut last_query = start;
    let mut last_input: Option<f64> = None;
    let mut next_battery = start;
    let mut stopping: Option<Stopping> = None;
    let mut buffer = vec![0u8; 1 << 16];
    let mut ticker = tokio::time::interval(TICK);
    ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);

    let request = match session.request() {
        Ok(request) => request,
        Err(error) => return from_error(error),
    };
    if let Err(end) = send(io, &request).await {
        return end;
    }

    // Begin a graceful stop; returns the end right away when there's nothing to wait for.
    macro_rules! begin_stop {
        ($reason:expr) => {{
            let reason = $reason;
            match session.stop() {
                Ok(bytes) if bytes.is_empty() => return reason,
                Ok(bytes) => {
                    if let Err(end) = send(io, &bytes).await {
                        return end;
                    }
                    stopping = Some(Stopping { until: clock.now() + STOP_WAIT, reason });
                }
                Err(_) => return reason,
            }
        }};
    }

    loop {
        tokio::select! {
            read = io.read(&mut buffer) => {
                let count = match read {
                    Ok(0) => return stopping.map_or(LinkEnd::StreamEnded, |stop| stop.reason),
                    Ok(count) => count,
                    Err(error) => return LinkEnd::Failed(format!("read from the band failed: {error}")),
                };
                let now = clock.now();
                last_read = now;
                let result = match session.feed(&buffer[..count], now) {
                    Ok(result) => result,
                    Err(error) => return from_error(error),
                };
                if let Err(end) = send(io, &result.outgoing).await {
                    return end;
                }
                for event in result.events {
                    match event {
                        Event::Connected => {
                            live = true;
                            deadline = f64::INFINITY;
                            last_input = Some(now);
                        }
                        Event::Heartbeat | Event::DataSeen => last_input = Some(now),
                        _ => {}
                    }
                    if stopping.is_none() {
                        let _ = events.send(LinkEvent::Band(event)).await;
                    }
                }
                if let Some(stop) = &stopping
                    && session.stop_acknowledged()
                {
                    return stop.reason.clone();
                }
            }
            _ = ticker.tick() => {
                let now = clock.now();
                if let Some(stop) = &stopping {
                    if now >= stop.until {
                        return stop.reason.clone();
                    }
                    continue;
                }
                if now >= deadline {
                    return LinkEnd::StartupTimeout;
                }
                for event in session.tick(now) {
                    let _ = events.send(LinkEvent::Band(event)).await;
                }
                let mut outgoing = Vec::new();
                if session.streams_enabled() && now >= next_battery {
                    next_battery = now + BATTERY_EVERY;
                    match session.query_battery_status(now) {
                        Ok(bytes) => outgoing.extend(bytes),
                        Err(error) => return from_error(error),
                    }
                }
                if session.streams_enabled() && now - last_read >= QUIET_QUERY && now - last_query >= QUIET_QUERY {
                    last_query = now;
                    match session.query_stream_state() {
                        Ok(bytes) => outgoing.extend(bytes),
                        Err(error) => return from_error(error),
                    }
                }
                if let Err(end) = send(io, &outgoing).await {
                    return end;
                }
                let silent = now - last_input.unwrap_or(start);
                let limit = if live { STALL_LIVE } else { STALL_BEFORE_INPUT };
                if (live || !ceremony) && silent > limit {
                    begin_stop!(LinkEnd::Stalled);
                }
            }
            command = commands.recv(), if stopping.is_none() => {
                let now = clock.now();
                let bytes = match command {
                    None | Some(LinkCommand::Stop) => {
                        begin_stop!(LinkEnd::Closed);
                        continue;
                    }
                    Some(LinkCommand::SetHand(hand)) => session.set_handedness(hand, now),
                    Some(LinkCommand::RawEmg(on)) => session.set_raw_emg_enabled(on, now),
                    Some(LinkCommand::ResumePairRequest { signature, receipt }) => {
                        session.ceremony_pair_request_completed(&signature, &receipt)
                    }
                    Some(LinkCommand::ResumePair { signature, receipt, device_public_key }) => {
                        match session.ceremony_pair_completed(&signature, &receipt, device_public_key.as_deref()) {
                            Ok(bytes) => {
                                // The band may commit as soon as it reads these bytes.
                                if let Some(identity) = session.pending_identity()
                                    && let Err(error) = persist_pending(&identity)
                                {
                                    return LinkEnd::Failed(format!("could not save the pending owner key: {error}"));
                                }
                                Ok(bytes)
                            }
                            Err(error) => Err(error),
                        }
                    }
                };
                match bytes {
                    Ok(bytes) => {
                        if let Err(end) = send(io, &bytes).await {
                            return end;
                        }
                    }
                    Err(BandError::Busy(message)) => {
                        let _ = events.send(LinkEvent::Refused(message)).await;
                    }
                    Err(error) => return from_error(error),
                }
            }
        }
    }
}
```

```rust
fn percent_encode(text: &str) -> String {
    text.bytes()
        .map(|byte| {
            if byte.is_ascii_alphanumeric() || b"-._~".contains(&byte) {
                (byte as char).to_string()
            } else {
                format!("%{byte:02X}")
            }
        })
        .collect()
}

/// `application/x-www-form-urlencoded` with every reserved character escaped
/// (a space is `%20`, a plus `%2B`), matching kinesis `MetaAuth.urlForm`.
pub fn url_form(fields: &[(&str, String)]) -> String {
    fields
        .iter()
        .map(|(name, value)| format!("{}={}", percent_encode(name), percent_encode(value)))
        .collect::<Vec<_>>()
        .join("&")
}

fn decode_signature(json: &Value) -> Result<Vec<u8>, PairError> {
    B64.decode(json.get("receipt_signature").and_then(Value::as_str).unwrap_or(""))
        .map_err(|_| PairError::Failed("the band claim service returned an unreadable signature. try again.".into()))
}

pub fn parse_pending(json: &Value) -> Result<Pending, PairError> {
    let receipt = json.get("pending_ownership_receipt").and_then(Value::as_str).filter(|r| !r.is_empty());
    let Some(receipt) = receipt else {
        return Err(PairError::Failed("the band claim service didn't return a receipt. try again.".into()));
    };
    Ok(Pending { signature: decode_signature(json)?, receipt: receipt.to_owned() })
}

fn device_key(json: &Value) -> Option<Vec<u8>> {
    let key = json.get("additional_data")?.get("device_ec_public_key")?.as_str()?;
    B64.decode(key).ok()
}

pub fn parse_final(json: &Value) -> Result<Final, PairError> {
    let receipt = json.get("final_ownership_receipt").and_then(Value::as_str).filter(|r| !r.is_empty());
    let Some(receipt) = receipt else {
        return Err(PairError::Failed("the band claim service didn't return a final receipt. try again.".into()));
    };

        device_key(json).or_else(|| serde_json::from_str::<Value>(receipt).ok().and_then(|inner| device_key(&inner)));

}

pub fn is_session_failure(status: u16, error: Option<&Value>) -> bool {

}

}

pub struct HttpPairClient {
    http: reqwest::Client,
    session: MetaSession,
}

impl HttpPairClient {
    pub fn new(session: MetaSession) -> Self {
        let http = reqwest::Client::builder().timeout(Duration::from_secs(30)).build().expect("HTTP client");
        Self { http, session }
    }

    async fn execute(&self, route: &str, fields: Vec<(&'static str, String)>) -> Result<Value, PairError> {
        let response = self
            .http
            .post(format!("{HOST}/{route}"))
            .header(reqwest::header::CONTENT_TYPE, "application/x-www-form-urlencoded")
            .header(reqwest::header::ACCEPT, "application/json")
            .body(url_form(&fields))
            .send()
            .await
            .map_err(|error| PairError::Failed(format!("the band claim service couldn't be reached: {error}")))?;
        let status = response.status().as_u16();
        let json: Option<Value> = response.json().await.ok();
        if is_session_failure(status, json.as_ref().and_then(|j| j.get("error"))) {
            return Err(PairError::SessionInvalid("your meta session expired. sign in to claim the band again.".into()));
        }
        let json = match json {
            Some(json) if status == 200 => json,
            _ => return Err(PairError::Failed(format!("the band claim service failed (http {status}). try again."))),
        };
        if let Some(error) = json.get("error") {
            let code = error.get("code").and_then(Value::as_i64).map_or("-".to_string(), |code| code.to_string());
            return Err(PairError::Failed(format!("the band claim service rejected the request (code {code}). try again.")));
        }
        Ok(json)
    }
```

```rust
//! Claiming a band for this computer: find it in pairing mode, pair at the
//! Bluetooth level, run the ownership ceremony with Meta's two HTTP steps and
//! save the new owner key — as pending before the band can commit, confirmed
//! once it does (kinesis `BandModel.pairBand` / `performCeremony`).

use std::time::Duration;

use band_core::ceremony::{CeremonyHttpRequest, OwnershipCeremony};
use band_core::events::Event;
use band_core::session::BandSession;
use tokio::sync::mpsc;

use crate::clock::Clock;

use crate::meta::{PairClient, PairError};
use crate::store::{BandRecord, Store};
use crate::transport::Transport;

#[derive(Debug, Clone, PartialEq)]
pub enum PairFailure {
    NoBandFound,
    /// The Meta session is gone or expired: sign in again.
    SessionExpired(String),
    /// The band belongs to another Meta account (`0x1042`).
    WrongAccount(String),
    Rejected(String),
    Failed(String),
}

impl std::fmt::Display for PairFailure {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            PairFailure::NoBandFound => f.write_str(
                "no band in pairing mode found. hold the band's button for 3 seconds until it flashes, then try again.",
            ),
            PairFailure::SessionExpired(message)
            | PairFailure::WrongAccount(message)
            | PairFailure::Rejected(message)
            | PairFailure::Failed(message) => f.write_str(message),
        }
    }
}

fn from_pair_error(error: PairError) -> PairFailure {
    match error {
        PairError::SessionInvalid(message) => PairFailure::SessionExpired(message),
        PairError::Failed(message) => PairFailure::Failed(message),
    }
}

/// Claim the band at `address` (or the first one found) for this computer.
pub async fn pair_band<T: Transport, C: PairClient>(
    transport: &T,
    client: &C,
    store: &Store,
    address: Option<&str>,
    clock: Clock,
    mut progress: impl FnMut(&str) + Send,
) -> Result<BandRecord, PairFailure> {
    progress("looking for a band in pairing mode");
    let found = transport.scan(Duration::from_secs(10)).await.map_err(PairFailure::Failed)?;
    let chosen = match address {
        Some(address) => found.into_iter().find(|band| band.address.eq_ignore_ascii_case(address)),
        None => found.into_iter().next(),
    }
    .ok_or(PairFailure::NoBandFound)?;
    let record = BandRecord { address: chosen.address, name: chosen.name };

    progress("preparing Bluetooth pairing");
    let _pairing = transport.prepare_pairing(&record.address).await.map_err(PairFailure::Failed)?;
    progress(&format!("connecting to {}", record.name));
    let io = transport.open(&record.address).await.map_err(PairFailure::Failed)?;

    let (commands, command_rx) = mpsc::channel(8);
    let (event_tx, mut events) = mpsc::channel(64);
    let pending_store = store.clone();
    let session = BandSession::new(None, Some(OwnershipCeremony::default()), false);
    let link = tokio::spawn(run_link(io, session, clock, command_rx, event_tx, move |identity| {
        pending_store.save_pending(identity)
    }));

    let mut failure: Option<PairFailure> = None;
    let mut saved = false;
    while let Some(event) = events.recv().await {
        let LinkEvent::Band(event) = event else { continue };
        match event {
            Event::CeremonyStage(stage) => progress(&stage),
            Event::CeremonyHttp(CeremonyHttpRequest::PairRequest(request)) => {
                progress("asking Meta to approve the claim");
                let next = match client.pair_request(&request).await {
                    Ok(pending) => LinkCommand::ResumePairRequest { signature: pending.signature, receipt: pending.receipt },
                    Err(error) => {
                        failure = Some(from_pair_error(error));
                        LinkCommand::Stop
                    }
                };
                let _ = commands.send(next).await;
            }
            Event::CeremonyHttp(CeremonyHttpRequest::Pair(pair)) => {
                progress("confirming the claim with Meta");
                let next = match client.pair(&pair).await {
                    Ok(done) => LinkCommand::ResumePair {
                        signature: done.signature,
                        receipt: done.receipt,
                        device_public_key: done.device_public_key,
                    },
                    Err(error) => {
                        failure = Some(from_pair_error(error));
                        LinkCommand::Stop
                    }
                };
                let _ = commands.send(next).await;
            }
            Event::CeremonyCompleted(identity) => match store.save_identity(&identity) {
                Ok(()) => {
                    saved = true;
                    progress("the band accepted this computer as its owner");
                }
                Err(error) => {
                    failure = Some(PairFailure::Failed(format!("could not save the owner key: {error}")));
                    let _ = commands.send(LinkCommand::Stop).await;
                }
            },
            Event::Connected => {
                progress("connected as the owner");
                let _ = commands.send(LinkCommand::Stop).await;
            }
            _ => {}
        }
    }
    let outcome = link.await.map_err(|error| PairFailure::Failed(format!("the pairing task failed: {error}")))?;
    if !saved
        && let Some(identity) = &outcome.enrollment
        && store.save_identity(identity).is_ok()
    {
        saved = true;
    }
    // A pending key means the band may already be ours: keep the band so a
    // later connect can try that key.
    if saved || store.load_pending().ok().flatten().is_some() {
        store
            .save_band(&record)
            .map_err(|error| PairFailure::Failed(format!("could not save the band record: {error}")))?;
    }
    if let Some(failure) = failure {
        return Err(failure);
    }
    if saved {
        return Ok(record);
    }
    Err(match outcome.end {
        LinkEnd::OwnershipRejected { code: 0x1042, message } => PairFailure::WrongAccount(message),
        LinkEnd::OwnershipRejected { message, .. } | LinkEnd::IdentityRejected(message) => PairFailure::Rejected(message),
        LinkEnd::Failed(message) => PairFailure::Failed(message),
        other => PairFailure::Failed(format!("the claim didn't finish: {other}")),
    })
}
```

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum HandChoice {
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
#[serde(tag = "cmd", rename_all = "kebab-case")]
pub enum Request {
    Status,
    /// A status line now and on every change.
    Subscribe,
    /// One line per raw sEMG frame (turn streaming on with `semg`).
    SubscribeSemg,
    Connect,
    Disconnect,

    Pair {

        address: Option<String>,
    },
    /// Forget the band: keys, record, Meta session and the Bluetooth bond.
    Forget,
}

#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
pub struct Response {
    pub ok: bool,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub status: Option<Status>,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub message: Option<String>,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub error: Option<String>,
    /// One raw sEMG payload, base64 (only on `subscribe-semg` streams).
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub semg: Option<String>,
}

impl Response {
    pub fn ok(status: Status) -> Self {
        Self { ok: true, status: Some(status), ..Self::default() }
    }

    pub fn message(status: Status, message: impl Into<String>) -> Self {
        Self { ok: true, status: Some(status), message: Some(message.into()), ..Self::default() }
    }

    pub fn error(message: impl Into<String>) -> Self {
        Self { ok: false, error: Some(message.into()), ..Self::default() }
    }
}

/// (`kinesis-dry-run.sock` for a dry run), else under `/tmp`.
pub fn socket_path(dry_run: bool) -> PathBuf {
    if let Some(path) = std::env::var_os("KINESIS_SOCKET") {
        return PathBuf::from(path);
    }
    let name = if dry_run { "kinesis-dry-run.sock" } else { "kinesis.sock" };
    match std::env::var_os("XDG_RUNTIME_DIR") {
        Some(dir) => PathBuf::from(dir).join(name),
        None => {
            let user = std::env::var("USER").unwrap_or_else(|_| "user".into());
            PathBuf::from(format!("/tmp/{user}-{name}"))
        }
    }
}

/// Bind the socket (0600). A live socket means another daemon runs; a stale file is replaced.
pub fn bind(path: &Path) -> io::Result<UnixListener> {
    if path.exists() {
        if std::os::unix::net::UnixStream::connect(path).is_ok() {
            return Err(io::Error::new(io::ErrorKind::AddrInUse, "kinesisd is already running"));
        }
        std::fs::remove_file(path)?;
    }
    let listener = UnixListener::bind(path)?;
    std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600))?;
    Ok(listener)
}

pub async fn serve(listener: UnixListener, handles: Handles) {
    loop {
        match listener.accept().await {
            Ok((stream, _)) => {
                let handles = handles.clone();
                tokio::spawn(async move {
                    if let Err(error) = connection(stream, handles).await {
                        tracing::debug!("control client: {error}");
                    }
                });
            }
            Err(error) => {
                tracing::warn!("control socket accept failed: {error}");
                tokio::time::sleep(Duration::from_millis(100)).await;
            }
        }
    }
}

async fn write_line<W: AsyncWrite + Unpin>(writer: &mut W, response: &Response) -> io::Result<()> {
    let mut line = serde_json::to_vec(response).map_err(io::Error::other)?;
    line.push(b'\n');
    writer.write_all(&line).await
}

async fn connection(stream: UnixStream, handles: Handles) -> io::Result<()> {
    let (read, mut write) = stream.into_split();
    let mut lines = BufReader::new(read).lines();
    while let Some(line) = lines.next_line().await? {
        if line.trim().is_empty() {
            continue;
        }
        let request = match serde_json::from_str::<Request>(&line) {
            Ok(request) => request,
            Err(error) => {
                write_line(&mut write, &Response::error(format!("bad request: {error}"))).await?;
                continue;
            }
        };
        match request {
            Request::Subscribe => {
                let mut status = handles.status.clone();
                loop {
                    let current = status.borrow_and_update().clone();
                    write_line(&mut write, &Response::ok(current)).await?;
                    if status.changed().await.is_err() {
                        return Ok(());
                    }
                }
            }
            Request::SubscribeSemg => {
                let mut frames = handles.semg.subscribe();
                loop {
                    match frames.recv().await {
                        Ok(frame) => {
                            let line = Response { ok: true, semg: Some(B64.encode(frame)), ..Response::default() };
                            write_line(&mut write, &line).await?;
                        }
                        Err(broadcast::error::RecvError::Lagged(_)) => continue,
                        Err(broadcast::error::RecvError::Closed) => return Ok(()),
                    }
                }
            }
            request => {
                let (reply, response) = oneshot::channel();
                if handles.calls.send((request, reply)).await.is_err() {
                    return Ok(());
                }
                let response = response.await.unwrap_or_else(|_| Response::error("kinesisd is shutting down"));
                write_line(&mut write, &response).await?;
            }
        }
    }
    Ok(())
}

/// One request, one response (for kinesisctl and tests).
pub async fn call(path: &Path, request: &Request) -> io::Result<Response> {
    let stream = UnixStream::connect(path).await?;
    let (read, mut write) = stream.into_split();
    let mut line = serde_json::to_vec(request).map_err(io::Error::other)?;
    line.push(b'\n');
    write.write_all(&line).await?;
    let mut lines = BufReader::new(read).lines();
    let answer = lines
        .next_line()
        .await?
        .ok_or_else(|| io::Error::new(io::ErrorKind::UnexpectedEof, "kinesisd closed the connection"))?;
    serde_json::from_str(&answer).map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))
}
```

```rust
use std::path::{Path, PathBuf};

pub struct Daemon<T: Transport, R: Runner, F: ClientFactory> {

fn modified(path: &Path) -> Option<SystemTime> {

}

async fn next_event(link: &mut Option<ActiveLink>) -> Option<LinkEvent> {
    match link {
        Some(link) => link.events.recv().await,
        None => std::future::pending().await,
    }
}

async fn join<T>(task: &mut Option<JoinHandle<T>>) -> Result<T, String> {
    match task {
        Some(task) => task.await.map_err(|error| error.to_string()),
        None => std::future::pending().await,
    }
}

impl<T: Transport, R: Runner, F: ClientFactory> Daemon<T, R, F> {
    pub fn new(transport: Arc<T>, runner: R, factory: F, store: Store, config_path: PathBuf) -> Self {
        let (config, config_error) = match Config::load(&config_path) {
            Ok(config) => (config, None),
            Err(error) => (Config::default(), Some(format!("config: {error}"))),
        };
        let config_stamp = modified(&config_path);
        let wants_connection = config.band.autostart;
        let band_name = store.load_band().ok().flatten().map(|band| band.name);
        let (status, _) = watch::channel(Status::default());
        let (semg, _) = broadcast::channel(256);
        let (progress_tx, progress_rx) = mpsc::unbounded_channel();
        let daemon = Self {
            transport,
            runner,
            factory: Arc::new(factory),
            store,
            config_path,
            config_stamp,
            config_error,
            controller: Controller::new(config),
            clock: Clock::new(),
            status,
            semg,
            wants_connection,
            retries: 0,
            reconnect_at: wants_connection.then_some(0.0),
            link: None,
            state: LinkState::Idle,
            error: None,
            pairing: None,
            pairing_stage: None,
            progress_tx,
            progress_rx,
            semg_wanted: false,
            semg_on: false,
            band_name,
        };
        daemon.publish();
        daemon
    }

    pub fn status(&self) -> watch::Receiver<Status> {
        self.status.subscribe()
    }

    pub fn semg(&self) -> broadcast::Sender<Vec<u8>> {
        self.semg.clone()
    }

    pub async fn run(mut self, mut calls: mpsc::Receiver<Call>, shutdown: impl Future<Output = ()>) {
        tokio::pin!(shutdown);
        let mut ticker = tokio::time::interval(TICK);
        ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
        loop {
            tokio::select! {
                _ = &mut shutdown => break,
                call = calls.recv() => match call {
                    Some((request, reply)) => {
                        let response = self.handle(request).await;
                        let _ = reply.send(response);
                    }
                    None => break,
                },
                event = next_event(&mut self.link) => match event {
                    Some(event) => self.on_link_event(event).await,
                    None => self.on_link_closed().await,
                },
                result = join(&mut self.pairing) => {
                    self.pairing = None;
                    self.on_pairing_end(result);
                }
                Some(stage) = self.progress_rx.recv() => self.pairing_stage = Some(stage),
                _ = ticker.tick() => self.on_tick().await,
            }
            self.publish();
        }
        self.stop_link().await;
        if let Some(pairing) = self.pairing.take() {
            pairing.abort();
        }
    }

    fn publish(&self) {
        let mut status = self.controller.status().clone();
        status.state = self.state;
        status.error = self.error.clone().or_else(|| self.config_error.clone());
        status.pairing = self.pairing_stage.clone();
        status.semg = self.semg_on;
        status.band = self.band_name.clone();
        self.status.send_if_modified(|current| {
            if *current == status {
                false
            } else {
                *current = status;
                true
            }
        });
    }

    async fn send_link(&self, command: LinkCommand) -> bool {
        match &self.link {
            Some(link) => link.commands.send(command).await.is_ok(),
            None => false,
        }
    }

    async fn apply(&mut self, command: Command) {
        match command {
            Command::Run(command) => {
                tracing::info!(%command, "action");
                self.runner.run(command);
            }
            Command::SetHand(hand) => {
                self.send_link(LinkCommand::SetHand(hand)).await;
            }
        }
    }

    async fn on_link_event(&mut self, event: LinkEvent) {
        let event = match event {
            LinkEvent::Refused(message) => {
                tracing::info!("band: {message}");
                return;
            }
            LinkEvent::Band(event) => event,
        };
        match &event {
            Event::Connected => {
                self.state = LinkState::Connected;
                self.retries = 0;
                self.error = None;
                tracing::info!("band connected");
                // The band accepted the key from an unconfirmed claim: it is ours now.
                if let Some(link) = &mut self.link
                    && link.pending
                {
                    link.pending = false;
                    if let Err(error) = self.store.promote_pending() {
                        tracing::warn!("could not confirm the pending owner key: {error}");
                    }
                }
            }
            Event::RawEmgFrame(frame) => {
                let _ = self.semg.send(frame.clone());
            }
            Event::RawEmgState(on) => self.semg_on = *on,
            Event::RawEmgFailure(message) | Event::HandednessFailure(message) => tracing::warn!("band: {message}"),
            _ => {}
        }
        let now = self.clock.now();
        for command in self.controller.on_event(&event, now) {
            self.apply(command).await;
        }
    }

    async fn on_link_closed(&mut self) {
        let Some(link) = self.link.take() else { return };
        let used_pending = link.pending;
        let outcome = link
            .task
            .await
            .unwrap_or_else(|error| LinkOutcome { end: LinkEnd::Failed(error.to_string()), enrollment: None });
        self.controller.on_disconnect();
        self.semg_on = false;
        self.state = LinkState::Idle;
        match &outcome.end {
            // The unconfirmed claim never committed: drop that key and retry with the confirmed one.
            LinkEnd::IdentityRejected(_) if used_pending && self.store.load_identity().ok().flatten().is_some() => {
                let _ = self.store.discard_pending();
                self.reconnect_at = Some(self.clock.now());
                return;
            }
            LinkEnd::IdentityRejected(message) => {
                self.error = Some(format!("{message} Run `kinesisctl pair` to claim the band again."));
                self.wants_connection = false;
            }
            LinkEnd::Closed => {}
            other => {
                tracing::warn!("band link ended: {other}");
                self.error = Some(other.to_string());
            }
        }
        if self.wants_connection {
            self.schedule_reconnect();
        }
    }

    fn schedule_reconnect(&mut self) {
        self.retries += 1;
        let delay = f64::from((2 * self.retries).min(10));
        self.reconnect_at = Some(self.clock.now() + delay);
    }

    async fn start_link(&mut self) {
        let band = match self.store.load_band() {
            Ok(Some(band)) => band,
            Ok(None) => {
                self.error = Some("no band paired yet — run `kinesisctl pair`".into());
                self.wants_connection = false;
                return;
            }
            Err(error) => {
                self.error = Some(format!("state: {error}"));
                self.wants_connection = false;
                return;
            }
        };
        let (identity, pending) = match self.store.connect_identity() {
            Ok(Some((identity, source))) => (Some(identity), source == IdentitySource::Pending),
            Ok(None) => (None, false),
            Err(error) => {
                self.error = Some(format!("state: {error}"));
                self.wants_connection = false;
                return;
            }
        };
        self.band_name = Some(band.name.clone());
        self.state = LinkState::Connecting;
        let (commands, command_rx) = mpsc::channel(16);
        let (event_tx, events) = mpsc::channel(256);
        let transport = self.transport.clone();
        let session = BandSession::new(identity, None, self.semg_wanted);
        let clock = self.clock;
        let task = tokio::spawn(async move {
            match transport.open(&band.address).await {
                Ok(io) => run_link(io, session, clock, command_rx, event_tx, |_| Ok(())).await,
                Err(error) => {
                    LinkOutcome { end: LinkEnd::Failed(format!("could not reach {}: {error}", band.name)), enrollment: None }
                }
            }
        });
        self.link = Some(ActiveLink { commands, events, task, pending });
    }

    async fn stop_link(&mut self) {
        let Some(link) = self.link.take() else { return };
        let _ = link.commands.send(LinkCommand::Stop).await;
        drop(link.commands);
        drop(link.events);
        let mut task = link.task;
        if tokio::time::timeout(Duration::from_secs(5), &mut task).await.is_err() {
            task.abort();
        }
        self.controller.on_disconnect();
        self.semg_on = false;
        self.state = LinkState::Idle;
    }

    async fn on_tick(&mut self) {
        let now = self.clock.now();
        if self.wants_connection
            && self.link.is_none()
            && self.pairing.is_none()
            && self.reconnect_at.is_some_and(|at| now >= at)
        {
            self.reconnect_at = None;
            self.start_link().await;
        }
        self.reload_config().await;
    }

    async fn reload_config(&mut self) {
        let stamp = modified(&self.config_path);
        if stamp == self.config_stamp {
            return;
        }
        self.config_stamp = stamp;
        match Config::load(&self.config_path) {
            Ok(config) => {
                self.config_error = None;
                tracing::info!("config reloaded");
                for command in self.controller.set_config(config) {
                    self.apply(command).await;
                }
            }
            Err(error) => {
                tracing::warn!("config: {error}");
                self.config_error = Some(format!("config: {error}"));
            }
        }
    }

    fn on_pairing_end(&mut self, result: Result<Result<BandRecord, PairFailure>, String>) {
        self.pairing_stage = None;
        self.state = LinkState::Idle;
        match result {
            Ok(Ok(record)) => {
                tracing::info!("paired with {}", record.name);
                self.band_name = Some(record.name);
                self.error = None;
                // The settling reconnect: connect again as the new owner.
                self.wants_connection = true;
                self.retries = 0;
                self.reconnect_at = Some(self.clock.now());
            }
            Ok(Err(PairFailure::SessionExpired(message))) => {
                let _ = self.store.delete_meta_session();
                self.error = Some(message);
            }
            Ok(Err(failure)) => self.error = Some(failure.to_string()),
            Err(error) => self.error = Some(format!("the pairing crashed: {error}")),
        }
    }

    async fn start_pairing(&mut self, address: Option<String>) -> Response {
        let session = match self.store.load_meta_session() {
            Ok(session) => session,
            Err(error) => return Response::error(format!("state: {error}")),
        };
        let Ok(client) = self.factory.client(session) else {
            return Response::error("sign in to Meta first (`kinesisctl pair` opens the sign-in window)");
        };
        self.wants_connection = false;
        self.reconnect_at = None;
        self.stop_link().await;
        self.state = LinkState::Pairing;
        self.error = None;
        self.pairing_stage = Some("starting".into());
        let transport = self.transport.clone();
        let store = self.store.clone();
        let clock = self.clock;
        let progress = self.progress_tx.clone();
        self.pairing = Some(tokio::spawn(async move {
            pair_band(&*transport, &client, &store, address.as_deref(), clock, move |stage: &str| {
                let _ = progress.send(stage.to_owned());
            })
            .await
        }));
        self.publish();
        Response::message(
            self.status.borrow().clone(),
            "put the band in pairing mode (hold its button for 3 seconds until it flashes)",
        )
    }

    async fn handle(&mut self, request: Request) -> Response {
        let now = self.clock.now();
        match request {
            Request::Status => {}
            Request::Connect => {
                if self.pairing.is_some() {
                    return Response::error("a pairing is running");
                }
                self.wants_connection = true;
                self.retries = 0;
                self.error = None;
                if self.link.is_none() {
                    self.reconnect_at = None;
                    self.start_link().await;
                }
            }
            Request::Disconnect => {
                self.wants_connection = false;
                self.reconnect_at = None;
                self.stop_link().await;
            }
            Request::Pause => self.controller.pause(),
            Request::Resume => self.controller.resume(now),
            Request::TogglePause => self.controller.toggle(now),
            Request::Dial { target } => self.controller.set_dial_target(target),
            Request::CycleDial => self.controller.cycle_dial(),
            Request::Hand { hand } => {
                if self.state != LinkState::Connected {
                    return Response::error("connect the band first");
                }
                let hand = match hand {
                    HandChoice::Left => Hand::Left,
                    HandChoice::Right => Hand::Right,
                };
                // Through the controller, so the config's hand doesn't undo the choice.
                for command in self.controller.request_hand(hand) {
                    self.apply(command).await;
                }
            }
            Request::Semg { on } => {
                self.semg_wanted = on;
                self.send_link(LinkCommand::RawEmg(on)).await;
            }
            Request::Pair { address } => {
                if self.pairing.is_some() {
                    return Response::error("a pairing is already running");
                }
                return self.start_pairing(address).await;
            }
            Request::Forget => {
                self.wants_connection = false;
                self.reconnect_at = None;
                self.stop_link().await;
                if let Ok(Some(band)) = self.store.load_band()
                    && let Err(error) = self.transport.forget(&band.address).await
                {
                    tracing::warn!("could not remove the Bluetooth bond: {error}");
                }
                if let Err(error) = self.store.forget() {
                    return Response::error(format!("state: {error}"));
                }
                self.band_name = None;
                self.error = None;
                self.publish();
                return Response::message(
                    self.status.borrow().clone(),
                    format!("forgot the band. factory reset it too ({FACTORY_RESET_HINT}) before pairing it anywhere else."),
                );
            }
            Request::Subscribe | Request::SubscribeSemg => {
                return Response::error("subscriptions are handled by the control socket");
            }
        }
        self.publish();
        Response::ok(self.status.borrow().clone())
    }
}
```

Implementation notes:
- `tokio::select!` builds all branch futures (disjoint `&mut` borrows of `self.link`, `self.pairing`, `self.progress_rx`) and drops them before running the chosen handler, so the handlers can use `&mut self`. If the compiler still objects, move the three receivers into locals owned by `run` and pass them to the handlers.
- `ClientFactory::client` returns `Err(_)` for "no session"; the daemon only needs to know it failed.
- `Store::connect_identity()` returns `Option<(EnrollmentIdentity, IdentitySource)>` and prefers a pending key (batch-A review fix). `IdentitySource` needs `PartialEq` for the comparison in `start_link`; derive it if missing. A pending key the band accepts is promoted on `Connected`; one it rejects (0x1043) is discarded and the confirmed key is tried at once.
- The test harness seeds the simulated band's owner with `transport.set_owner(&identity)` because the simulated band now only trusts its owner, like a real one.
- `Controller::request_hand` (batch-B review fix) sets a runtime hand override and returns the `SetHand` command when the band's hand is confirmed.
- The hand test relies on `Controller` updating `status.hand` from the band's confirmed read-back after the write (the session re-reads the hand after every write).

```rust
//! The real transport: BlueZ via `bluer`.
//!
//! The band advertises a name starting "Meta Band". Its GATT service `FEB8`
//! holds characteristic `2d41da7c-…` whose value is the input channel's L2CAP
//! PSM as little-endian u16 (observed `ff 00` = 255). Reading it requires an
//! encrypted link, so the first read makes BlueZ pair — during `pair` our
//! scoped agent accepts that one request. The input channel is an LE
//! credit-based L2CAP stream on that PSM (kinesis `BandConnection.swift`).

use std::collections::HashMap;
use std::time::Duration;

use bluer::agent::{Agent, AgentHandle, ReqError};
use bluer::l2cap::{SocketAddr, Stream};
use bluer::{Adapter, AdapterEvent, Address, Device, DiscoveryFilter, DiscoveryTransport, Session, Uuid};
use futures::StreamExt;

use crate::transport::{Found, Transport};

const BAND_SERVICE: Uuid = Uuid::from_u128(0x0000feb8_0000_1000_8000_00805f9b34fb);
const PSM_CHARACTERISTIC: Uuid = Uuid::from_u128(0x2d41da7c_82b6_42aa_b34e_e2e01df8cc1a);

fn describe(error: bluer::Error) -> String {
    error.to_string()
}

fn parse(address: &str) -> Result<Address, String> {
    address.parse().map_err(|_| format!("not a Bluetooth address: {address}"))
}

pub struct BluezTransport {
    session: Session,
    adapter: Adapter,
}

impl BluezTransport {
    pub async fn new() -> Result<Self, String> {
        let session = Session::new().await.map_err(describe)?;
        let adapter = session.default_adapter().await.map_err(describe)?;
        adapter.set_powered(true).await.map_err(describe)?;
        Ok(Self { session, adapter })
    }
}

async fn read_psm(device: &Device) -> Result<u16, String> {
    for service in device.services().await.map_err(describe)? {
        if service.uuid().await.map_err(describe)? != BAND_SERVICE {
            continue;
        }
        for characteristic in service.characteristics().await.map_err(describe)? {
            if characteristic.uuid().await.map_err(describe)? != PSM_CHARACTERISTIC {
                continue;
            }
            let value = characteristic.read().await.map_err(|error| {
                format!("reading the band's input channel failed ({error}); if the band was factory reset, run `kinesisctl pair` again")
            })?;
            return match value.as_slice() {
                [low, high] => Ok(u16::from_le_bytes([*low, *high])),
                _ => Err("unsupported band input channel".into()),
            };
        }
    }
    Err("this device has no Neural Band input service".into())
}

impl Transport for BluezTransport {
    type Io = Stream;
    type PairingGuard = AgentHandle;

    async fn scan(&self, timeout: Duration) -> Result<Vec<Found>, String> {
        let filter = DiscoveryFilter { transport: DiscoveryTransport::Le, ..Default::default() };
        self.adapter.set_discovery_filter(filter).await.map_err(describe)?;
        let events = self.adapter.discover_devices().await.map_err(describe)?;
        let mut events = std::pin::pin!(events);
        let mut found: HashMap<Address, Found> = HashMap::new();
        let deadline = tokio::time::sleep(timeout);
        let mut deadline = std::pin::pin!(deadline);
        loop {
            tokio::select! {
                _ = &mut deadline => break,
                event = events.next() => match event {
                    Some(AdapterEvent::DeviceAdded(address)) => {
                        let Ok(device) = self.adapter.device(address) else { continue };
                        if let Ok(Some(name)) = device.name().await
                            && name.to_lowercase().starts_with("meta band")
                        {
                            let rssi = device.rssi().await.ok().flatten();
                            found.insert(address, Found { address: address.to_string(), name, rssi });
                        }
                    }
                    Some(_) => {}
                    None => break,
                },
            }
        }
        let mut found: Vec<Found> = found.into_values().collect();
        found.sort_by_key(|band| std::cmp::Reverse(band.rssi.unwrap_or(i16::MIN)));
        Ok(found)
    }

    async fn prepare_pairing(&self, address: &str) -> Result<AgentHandle, String> {
        let target = parse(address)?;
        // A factory reset gives the band a new identity key; a stale bond would make pairing fail.
        if let Ok(device) = self.adapter.device(target)
            && device.is_paired().await.unwrap_or(false)
        {
            self.adapter.remove_device(target).await.map_err(describe)?;
        }
        let only_the_band = move |device: Address| if device == target { Ok(()) } else { Err(ReqError::Rejected) };
        let agent = Agent {
            request_default: true,
            request_confirmation: Some(Box::new(move |request| Box::pin(async move { only_the_band(request.device) }))),
            request_authorization: Some(Box::new(move |request| Box::pin(async move { only_the_band(request.device) }))),
            authorize_service: Some(Box::new(move |request| Box::pin(async move { only_the_band(request.device) }))),
            ..Default::default()
        };
        self.session.register_agent(agent).await.map_err(describe)
    }

    async fn open(&self, address: &str) -> Result<Stream, String> {
        let target = parse(address)?;
        let device = self.adapter.device(target).map_err(describe)?;
        if !device.is_connected().await.map_err(describe)? {
            tokio::time::timeout(Duration::from_secs(20), device.connect())
                .await
                .map_err(|_| "the band didn't answer the connection (is it nearby and awake?)".to_string())?
                .map_err(describe)?;
        }
        // The first read may wait on Bluetooth pairing.
        let psm = tokio::time::timeout(Duration::from_secs(45), read_psm(&device))
            .await
            .map_err(|_| "the band didn't open its input channel in time".to_string())??;
        let kind = device.address_type().await.map_err(describe)?;
        tokio::time::timeout(Duration::from_secs(20), Stream::connect(SocketAddr::new(target, kind, psm)))
            .await
            .map_err(|_| "opening the band's input channel timed out".to_string())?
            .map_err(|error| format!("opening the band's input channel failed: {error}"))
    }

    async fn forget(&self, address: &str) -> Result<(), String> {
        match self.adapter.remove_device(parse(address)?).await {
            Ok(()) => Ok(()),
            Err(error) if error.kind == bluer::ErrorKind::NotFound => Ok(()),
            Err(error) => Err(describe(error)),
        }
    }
}
```

Notes for the implementer (API drift is likely; adapt, don't redesign):
- If `bluer`'s agent callback types differ (e.g. `RequestConfirmationFn` signature, `ReqError` variants), keep the rule "accept only `target`, reject everything else".
- If `bluer::ErrorKind::NotFound` doesn't exist, match whatever "device does not exist" maps to, or ignore any error from `remove_device` in `forget` with a `tracing::debug!`.
- If the `scan` future isn't `Send` because the discovery stream isn't, collect inside `tokio::task::spawn_local` is not available here — instead run the discovery loop in a `tokio::spawn`ed task that owns a cloned `Adapter` and returns the `Vec<Found>`.
- L2CAP security level: the stream connects over the bonded, encrypted LE link. If hardware day shows the band refusing the channel, try `Socket::new_stream()` + `set_security(Security { level: SecurityLevel::Medium, key_size: 0 })` + `connect`; record the finding.

`kinesisd/tests/binary.rs`:
```rust
use std::path::Path;
use std::process::{Child, Command, Stdio};
use std::time::{Duration, Instant};

use kinesisd::ipc::{self, Request, Response};
use kinesisd::status::LinkState;

struct Daemon(Child);

impl Drop for Daemon {
    fn drop(&mut self) {
        let _ = self.0.kill();
    }
}

async fn call(socket: &Path, request: Request) -> Response {
    ipc::call(socket, &request).await.unwrap()
}

async fn wait_until(socket: &Path, what: &str, until: impl Fn(&Response) -> bool) -> Response {
    let deadline = Instant::now() + Duration::from_secs(30);
    loop {
        if let Ok(response) = ipc::call(socket, &Request::Status).await
            && until(&response)
        {
            return response;
        }
        assert!(Instant::now() < deadline, "timed out waiting for {what}");
        tokio::time::sleep(Duration::from_millis(100)).await;
    }
}

#[tokio::test]
async fn the_dry_run_daemon_pairs_connects_and_forgets() {
    let dir = tempfile::tempdir().unwrap();
    let socket = dir.path().join("kinesis.sock");
    let child = Command::new(env!("CARGO_BIN_EXE_kinesisd"))
        .arg("--dry-run")
        .env("KINESIS_SOCKET", &socket)
        .env("XDG_STATE_HOME", dir.path().join("state"))
        .env("XDG_CONFIG_HOME", dir.path().join("config"))
        .env("KINESIS_LOG", "warn")
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .unwrap();
    let mut daemon = Daemon(child);

    let first = wait_until(&socket, "the unpaired error", |r| r.status.as_ref().is_some_and(|s| s.error.is_some())).await;
    assert!(first.status.unwrap().error.unwrap().contains("kinesisctl pair"));

    let paired = call(&socket, Request::Pair { address: None }).await;
    assert!(paired.ok, "{paired:?}");
    let connected = wait_until(&socket, "the owner connection", |r| {
        r.status.as_ref().is_some_and(|s| s.state == LinkState::Connected)
    })
    .await;
    assert_eq!(connected.status.unwrap().band.as_deref(), Some("Meta Band (simulated)"));
    assert!(dir.path().join("state/kinesis-dry-run/owner.key").exists());

    let forgot = call(&socket, Request::Forget).await;
    assert!(forgot.ok);
    assert!(!dir.path().join("state/kinesis-dry-run/owner.key").exists());

    // SIGTERM shuts it down cleanly and removes the socket.
    Command::new("kill").arg(daemon.0.id().to_string()).status().unwrap();
    let status = daemon.0.wait().unwrap();
    assert!(status.success(), "{status:?}");
    assert!(!socket.exists());
}
```

### Control socket contract (status v1)

One JSON object per line on `$XDG_RUNTIME_DIR/kinesis.sock` (`KINESIS_SOCKET` overrides).

Requests (`cmd` is kebab-case):

```json
{"cmd":"status"}            {"cmd":"subscribe"}          {"cmd":"subscribe-semg"}
{"cmd":"connect"}           {"cmd":"disconnect"}
{"cmd":"pause"}             {"cmd":"resume"}             {"cmd":"toggle-pause"}
{"cmd":"dial","target":"volume|brightness|off"}          {"cmd":"cycle-dial"}
{"cmd":"hand","hand":"left|right"}                       {"cmd":"semg","on":true}
{"cmd":"pair"}              {"cmd":"pair","address":"AA:BB:CC:DD:EE:FF"}
{"cmd":"forget"}
```

Responses (absent fields are omitted; `subscribe` sends one per status change, `subscribe-semg` one per frame):

```json
{"ok":true,"status":{…},"message":"optional text"}
{"ok":false,"error":"why"}
{"ok":true,"semg":"<base64 frame>"}
```
